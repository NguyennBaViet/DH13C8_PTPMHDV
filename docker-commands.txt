━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. SETUP & START
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Copy .env from example (first time only):
  cp .env.example .env
  (Update values in .env as needed)

Build all images:
  docker-compose build

Start all services:
  docker-compose up -d

Start with logs visible:
  docker-compose up

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
2. MONITORING & DEBUGGING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

View all running services:
  docker-compose ps

View logs (all services):
  docker-compose logs -f

View logs (specific service):
  docker-compose logs -f auth-service
  docker-compose logs -f payment-noti-service
  docker-compose logs -f api-gateway

View logs (last 100 lines):
  docker-compose logs --tail=100 auth-service

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
3. STOP & CLEANUP
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Stop all services (keep volumes):
  docker-compose down

Stop and remove volumes (reset database):
  docker-compose down -v

Restart specific service:
  docker-compose restart auth-service

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
4. DATABASE OPERATIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Connect to MySQL:
  docker exec -it khachsan-mysql mysql -u root -p khachsan

Check database tables:
  docker exec -it khachsan-mysql mysql -u root -p khachsan -e "SHOW TABLES;"

Backup database:
  docker exec khachsan-mysql mysqldump -u root -p khachsan > backup.sql

5. REBUILD & UPDATE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Rebuild specific service:
  docker-compose build auth-service
  docker-compose up -d auth-service

Rebuild all and restart:
  docker-compose down
  docker-compose build
  docker-compose up -d

  curl http://localhost:8022/health && \
  curl http://localhost:8023/health && \
  curl http://localhost:8024/health && \
  curl http://localhost:8025/health

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  docker logs khachsan-auth-service

8. COMMON ISSUES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Port already in use:
  docker-compose down -v && docker system prune -f

MySQL connection refused:
  docker-compose logs mysql
  Ensure DB_PASSWORD matches MYSQL_PASSWORD in .env

Service won't start:
  Check logs: docker-compose logs <service>
  Verify .env configuration
  Check Docker disk space: docker system df
